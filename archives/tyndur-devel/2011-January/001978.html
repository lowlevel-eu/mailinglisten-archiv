+#include <ctype.h>
+
+int isprint(int c)
+{
+    return (c >= 0x20);
+}
+
+int iscntrl(int c)
+{
+    return (c < 0x20);
+}
+
+int isspace(int c)
+{
+    return (c == ' ' || c== '\n' || c == '\t' || c == '\r');
+}
+
+int isblank(int c)
+{
+    return (c == ' ' ||c == '\t');
+}
+
+int isdigit(int c)
+{
+    return (c >= '0' && c <= '9');
+}
+
+int isalpha(int c)
+{
+    return ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'));
+}
+
+int isalnum(int c)
+{
+    return (isdigit(c) || isalpha(c));
+}
+
+int isxdigit(int c)
+{
+    return (isdigit(c) || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F'));
+}
+
+int isupper(int c)
+{
+    return (c >= 'A' && c <= 'Z');
+}
+
+int islower(int c)
+{
+    return (c >= 'a' && c <= 'z');
+}
+
+int ispunct(int c)
+{
+    return (isprint(c) && !isspace(c) && !isalpha(c));
+}
+
+int isgraph(int c)
+{
+    return (isprint(c) && !isspace(c));
+}
+
+int tolower(int c)
+{
+    return isupper(c) ? c + ('a' - 'A') : c;
+}
+
+int toupper(int c)
+{
+    return islower(c) ? c - ('a' - 'A') : c;
+}
