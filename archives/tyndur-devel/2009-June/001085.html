+#include <stdlib.h>
+
+void *bsearch(const void *key, const void *base, size_t nel,
+    size_t width, int (*compar)(const void *, const void *))
+{
+    size_t i;
+    const void* obj;
+
+    for (i = 0; i < nel; i++) {
+        obj = base + (width * i);
+        if (compar(key, obj) == 0) {
+            return (void*) obj;
+        }
+    }
+
+    return NULL;
+}
