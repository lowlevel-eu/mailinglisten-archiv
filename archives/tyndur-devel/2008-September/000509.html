void list_merge_into(list_t* dest, list_t* source)
{
    struct list_node* node;

    if (!dest || !source) {
        return;
    }

    node = list_get_node_at(source, source->size -1);
    node->next = dest->anchor;
    dest->size += source->size;

    free(source);
}
