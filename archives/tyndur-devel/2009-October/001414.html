+include toolchain
+include lutils
+
+DESCRIPTION="Ein Musikabspielprogramm"
+HOMEPAGE="http://lowlevel.brainsware.org/wiki/index.php/LOST/Pakete/player"
+SRC_URI="http://www.xanclic.bplaced.net/${P}.tar.bz2"
+
+LICENSE="MIT"
+
+BUILD_DEPEND="=libogg-1.1.4 =libvorbis-1.2.3"
+RUN_DEPEND=""
+
+MY_S="${S}/${P}"
+
+src_unpack() {
+lunpack ${P}.tar.bz2
+}
+
+src_compile() {
+	cd ${MY_S}
+
+	lmake CC=i386-pc-tyndur-gcc || die "make failed"
+	cd ..
+}
+
+src_install() {
+	cd ${MY_S}
+
+	lmake install INSTALL_TO="${D}${R}" || die "make install failed"
+	rm -rf ${D}${R}/share
+}
