+#include <math.h>
+#include <errno.h>
+
+double fmod(double x, double y)
+{
+    double res;
+
+    if (y == 0.0) {
+        errno = EDOM;
+        return NAN;
+    }
+
+    asm("fprem" : "=t" (res) : "0" (x), "u" (y));
+    return res;
+}
+
+float fmodf(float x, float y)
+{
+    float res;
+
+    if (y == 0.0) {
+        errno = EDOM;
+        return NAN;
+    }
+
+    asm("fprem" : "=t" (res) : "0" (x), "u" (y));
+    return res;
+}
+
+long double fmodl(long double x, long double y)
+{
+    long double res;
+
+    if (y == 0.0) {
+        errno = EDOM;
+        return NAN;
+    }
+
+    asm("fprem" : "=t" (res) : "0" (x), "u" (y));
+    return res;
+}
