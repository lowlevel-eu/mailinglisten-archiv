+#!/usr/bin/python
+# -*- coding: UTF-8 -*-
+
+# Eine leere Datei mit einer bestimmten Größe erzeugen
+# Parameter:
+#   1. Dateiname
+#   2. Dateigröße
+import sys
+
+def empty_file(filename, size):
+    f = open(filename,"wb")
+    if not f:
+        sys.exit(1)
+
+    f.seek(size - 1)
+    f.write('\0')
+    f.close()
+
+if __name__ == "__main__":
+    if len(sys.argv[1:]) != 2:
+        print sys.stderr, "using: %s <filename> <size>" % sys.argv[0]
+        sys.exit(1)
+
+    empty_file(str(sys.argv[1]), int(sys.argv[2]))
