+char* format_size(size_t size, bool human_readable)
+{
+    char* retval = NULL;
+    if (!human_readable) {
+        asprintf(&retval, "% 6d", size);
+    } else {
+        int unit = 0;
+        char units[] = " KMGT";
+        while (size > 1024 && units[unit]) {
+            size /= 1024;
+            unit++;
+        }
+        if (unit == 0) {
+            asprintf(&retval, "% 6d", size);
+        } else {
+            asprintf(&retval, " % 4d%c", size, units[unit]);
+        }
+    }
+    return retval;
+}
