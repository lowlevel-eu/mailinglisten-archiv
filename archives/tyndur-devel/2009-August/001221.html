+double ldexp(double x, int exp)
+{
+    double res = x;
+    int exponent;
+    int i;
+    if (!exp) {
+        if (!x) {
+            errno = EDOM;
+            return NAN;
+        }
+        return 1;
+    }
+    exponent = exp;
+    if (exp < 0) {
+        exponent = -exp;
+    }
+    for (i = 1; i < exponent; i++) {
+        res *= x;
+    }
+
+    if (exp < 0) {
+        return 1 / res;
+    }
+    
+    return res;
+}

+double atan2(double x, double y) {
+    double res;
+    if (x < 0) {
+        res = (pi - atan(y / x)) * sgn(y);
