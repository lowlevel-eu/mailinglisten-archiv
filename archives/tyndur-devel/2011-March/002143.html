+double cosh(double x) {
+    if (x == NAN) {
+        return NAN;
+    }
+    double res = (exp(x) + exp(-x)) / 2.0;
+    if (((res == INFINITY) || (res == -INFINITY))
+        && (x != INFINITY) && (x != - INFINITY))
+    {
+        errno = ERANGE;
+        return HUGE_VAL;
+    }
+    return res;
+}

+double sinh(double x) {
+    if (x == NAN) {
+        return NAN;
+    }
+    double res = (exp(x) - exp(-x)) / 2.0;
+    if (((res == INFINITY) || (res == -INFINITY))
+        && (x != INFINITY) && (x != - INFINITY))
+    {
+        errno = ERANGE;
+        if (x > 0.0) {
+            return HUGE_VAL;
+        } else {
+            return -HUGE_VAL;
+        }
+    }
+    return res;
+}
+
+float sinhf(float x) {
+    if (x == NAN) {
+        return NAN;
+    }
+    float res = (expf(x) - expf(-x)) / 2.0f;
+    if (((res == INFINITY) || (res == -INFINITY))
+        && (x != INFINITY) && (x != - INFINITY))
+    {
+        errno = ERANGE;
+        if (x > 0.0f) {
+            return HUGE_VALF;
+        } else {
+            return -HUGE_VALF;
+        }
+    }
+    return res;
+}

+double tanh(double x) {
+    if (x == NAN) {
+        return NAN;
+    } else if (x == INFINITY) {
+        return 1.0;
+    } else if (x == -INFINITY) {
+       return -1.0;
+    }
+    return sinh(x) / cosh(x);
+}

+       return -1.0f;
