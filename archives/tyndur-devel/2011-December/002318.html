+#include <stdlib.h>
+#include <stdio.h>
+#include <syscall.h>
+
+static void sync_display_usage(void);
+
+#ifdef CONFIG_SHELL_BUILTIN_ONLY
+    int shell_command_sync(int argc, char* argv[], const char* args)
+#else
+    int main(int argc, char* argv[])
+#endif
+{
+    if (argc != 1) {
+        sync_display_usage();
+        return -1;
+    }
+
+    lio_sync_all();
+    return EXIT_SUCCESS;
+}
+
+static void sync_display_usage()
+{
+    puts("\nAufruf: sync");
+}
