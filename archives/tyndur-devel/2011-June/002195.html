+cd fasm

+    patch source/formats.inc ${F}/formats.patch || die "could not patch formats.inc"

+src_compile() {
+	cd fasm
+	fasm source/libc/fasm.asm fasm.o
+	i386-pc-tyndur-gcc fasm.o -o fasm || die "linking failed"
+	cd ..
+}
+
+src_install() {
+	cd fasm
+	install -d ${D}${R}/bin
+	cp fasm ${D}${R}/bin || die "could not install fasm"
+}
